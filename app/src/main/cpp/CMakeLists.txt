cmake_minimum_required(VERSION 3.22.1)
project(zero_voice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(whispercpp GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG a8d002cfd879315632a579e73f0148d06959de36 GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(whispercpp)
add_library(zero_whisper SHARED bridge.cpp)
target_link_libraries(zero_whisper PRIVATE whisper)
